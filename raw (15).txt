local function StartCleanEngine()

            end
            return getgenv()[k]
        end
    })

    -- اصطياد loadstring
    pcall(function()
        local rawLoad = getgenv().loadstring or loadstring
        if rawLoad then
            OriginalHooks.loadstring = rawLoad
            getgenv().loadstring = function(code, chunk)
                local tag = "LOADSTRING"
                local l = string.lower(tostring(code))
                if string.find(l, "github") then tag = "GITHUB"

    end)

    -- اصطياد request

                elseif string.find(lu, "pastebin") then tag = "PASTEBIN"
